==============================================================================
02_rollback: Rollback, savepoints and a commit that fails
==============================================================================

-- PART 1. Someone tries to sell 50 yards of velvet when 20 are on the
-- shelf. The order and its line are written before the stock update fails.

[A] BEGIN;
        BEGIN
[A] INSERT INTO customer_order (customer_id, taken_by, notes)
    VALUES (:customer, :staff, 'Portfolio 3 exercise')
    RETURNING order_id \gset
        INSERT 0 1
[A] INSERT INTO order_item (order_id, item_kind, material_id, quantity, unit_price)
    VALUES (:order_id, 'material', :velvet, 50, 8000);
        INSERT 0 1
[A] UPDATE material SET qty_on_hand = qty_on_hand - 50 WHERE material_id = :velvet;
        ERROR:  new row for relation "material" violates check constraint "material_qty_on_hand_check"
        DETAIL:  Failing row contains (53, Royal blue velvet, fabric, yard, 8000.00, -30.00, 5.00).

-- The transaction is now marked as failed. PostgreSQL refuses any more
-- work in it, and answers COMMIT with ROLLBACK.

[A] SELECT qty_on_hand FROM material WHERE material_id = :velvet;
        ERROR:  current transaction is aborted, commands ignored until end of transaction block
[A] COMMIT;
        ROLLBACK

-- The order and line written before the error are gone as well.

[A] SELECT count(*) AS exercise_orders FROM customer_order WHERE notes = 'Portfolio 3 exercise';
         exercise_orders 
        -----------------
                       0
        (1 row)

-- PART 2. An order for 4 yards of lining and 50 yards of velvet. The
-- velvet fails again, but a savepoint lets the lining go through.

[A] BEGIN;
        BEGIN
[A] INSERT INTO customer_order (customer_id, taken_by, notes)
    VALUES (:customer, :staff, 'Portfolio 3 exercise')
    RETURNING order_id \gset
        INSERT 0 1
[A] INSERT INTO order_item (order_id, item_kind, material_id, quantity, unit_price)
    VALUES (:order_id, 'material', :lining, 4, 1500)
    RETURNING order_item_id \gset
        INSERT 0 1
[A] INSERT INTO stock_movement (material_id, movement_type, quantity, order_item_id, recorded_by)
    VALUES (:lining, 'sale', -4, :order_item_id, :staff);
        INSERT 0 1
[A] UPDATE material SET qty_on_hand = qty_on_hand - 4 WHERE material_id = :lining;
        UPDATE 1
[A] SAVEPOINT velvet_line;
        SAVEPOINT
[A] UPDATE material SET qty_on_hand = qty_on_hand - 50 WHERE material_id = :velvet;
        ERROR:  new row for relation "material" violates check constraint "material_qty_on_hand_check"
        DETAIL:  Failing row contains (53, Royal blue velvet, fabric, yard, 8000.00, -30.00, 5.00).
[A] ROLLBACK TO SAVEPOINT velvet_line;
        ROLLBACK
[A] COMMIT;
        COMMIT

-- Only the work after the savepoint was undone.

[A] SELECT name, qty_on_hand FROM material WHERE material_id IN (:velvet, :lining) ORDER BY name;
                name         | qty_on_hand 
        ---------------------+-------------
         Royal blue velvet   |       20.00
         Silver satin lining |       26.00
        (2 rows)

-- PART 3. A journal entry is keyed in with a mistake: 40,000 debited but
-- 4,000 credited. Both lines are accepted. The rule that debits equal
-- credits is checked at COMMIT, so it is the COMMIT that fails.

[A] BEGIN;
        BEGIN
[A] INSERT INTO journal_entry (description, posted_by)
    VALUES ('P3 exercise: mistyped entry', :staff)
    RETURNING entry_id \gset
        INSERT 0 1
[A] INSERT INTO journal_line (entry_id, account_id, debit, credit) VALUES
    (:entry_id, (SELECT account_id FROM account WHERE code = '1000'), 40000, 0),
    (:entry_id, (SELECT account_id FROM account WHERE code = '4020'), 0, 4000);
        INSERT 0 2
[A] COMMIT;
        ERROR:  journal entry 274959 is not balanced (debits - credits = 36000.00)
        CONTEXT:  PL/pgSQL function check_entry_balanced() line 18 at RAISE
[A] SELECT count(*) AS mistyped_entries FROM journal_entry WHERE description = 'P3 exercise: mistyped entry';
         mistyped_entries 
        ------------------
                        0
        (1 row)
