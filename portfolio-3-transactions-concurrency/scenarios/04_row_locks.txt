title: Two fixes - a row lock, and an update the database computes itself

# PART 1. The same two sales of 12 yards from a bolt of 20. This time each
# session reads the row with FOR UPDATE, which locks it until the
# transaction ends.

A: BEGIN;
A: SELECT qty_on_hand FROM material WHERE material_id = :velvet FOR UPDATE;
B: BEGIN;
B: SELECT qty_on_hand FROM material WHERE material_id = :velvet FOR UPDATE;

# B has to wait. A finishes its sale.

A: UPDATE material SET qty_on_hand = 8 WHERE material_id = :velvet;
A: INSERT INTO stock_movement (material_id, movement_type, quantity, recorded_by)
   VALUES (:velvet, 'sale', -12, :staff);
A: COMMIT;

# B's read returns the committed figure, 8, not the 20 it would have seen
# earlier. Its program can now see there is not enough and stop.

B: ROLLBACK;

# PART 2. Eight yards are left. Two sessions each try to sell 6. Neither
# reads first: the subtraction is written into the UPDATE itself.

A: BEGIN;
A: UPDATE material SET qty_on_hand = qty_on_hand - 6 WHERE material_id = :velvet;
B: BEGIN;
B: UPDATE material SET qty_on_hand = qty_on_hand - 6 WHERE material_id = :velvet;
A: INSERT INTO stock_movement (material_id, movement_type, quantity, recorded_by)
   VALUES (:velvet, 'sale', -6, :staff);
A: COMMIT;

# B's UPDATE waited for A's lock, then ran against the new value, 2. The
# CHECK constraint from Portfolio 1 refuses to let stock go below zero.

B: ROLLBACK;
A: SELECT m.qty_on_hand AS shelf_count,
          (SELECT sum(quantity) FROM stock_movement WHERE material_id = m.material_id) AS log_total
   FROM material m
   WHERE m.material_id = :velvet;
